### T-25-04-02: Playwright cross-user-leak.spec.ts

- Tạo `sources/frontend/tests/e2e/cross-user-leak.spec.ts`:
  ```typescript
  import { test, expect } from '@playwright/test';

  test('cross-user order leak via forged X-User-Id is blocked at gateway', async ({ request }) => {
    // 1. Login Alice
    const aliceLogin = await request.post('/api/users/auth/login', { data: { username: 'alice', password: '...' } });
    const aliceToken = (await aliceLogin.json()).data.token;
    const aliceId = decodeJwtSub(aliceToken);

    // 2. Login Bob
    const bobLogin = await request.post('/api/users/auth/login', { data: { username: 'bob', password: '...' } });
    const bobToken = (await bobLogin.json()).data.token;

    // 3. Bob attempts to list Alice's orders via forged header
    const leakAttempt = await request.get('/api/orders', {
      headers: {
        'Authorization': `Bearer ${bobToken}`,
        'X-User-Id': aliceId  // forged
      }
    });
    expect(leakAttempt.status()).toBe(200);
    const orders = (await leakAttempt.json()).data;
    // All orders must belong to Bob, NOT Alice
    for (const o of orders) {
      expect(o.userId).not.toBe(aliceId);
    }
  });

  test('protected endpoint without bearer returns 401', async ({ request }) => {
    const r = await request.get('/api/orders');
    expect(r.status()).toBe(401);
    expect((await r.json()).code).toBe('AUTH_TOKEN_MISSING');
  });
  ```
- Helper `decodeJwtSub(token)`: base64 decode middle segment, parse JSON, return `sub`.
- Seed user `alice` + `bob` phải tồn tại trong user-svc dev seed. Nếu chưa có → tạo migration thêm 2 user qua V100 hoặc dùng register API trong test setup (`beforeAll`).
